# ===========================================================================
# 真实语音引擎依赖（§5.3 音素级评测四步链路）

# 为什么单独成文件：`requirements.txt` 只装核心依赖，让 CI 与本地开发能在
# 零 GPU、零模型下载的机器上跑通全部测试（当前 `SPEECH_USE_FAKE=true`）。
# 部署真实评分能力时才安装本文件：
#
#     pip install -r requirements.txt -r requirements-pipeline.txt

# 系统级依赖（镜像里用 apt 安装，见 Dockerfile）：
#   ffmpeg     —— 链路第 0 步：webm/opus → 16kHz 单声道 wav（/speech/v1/transcode）
#   espeak-ng  —— 链路步骤③：预期文本 → IPA 音素序列
# ===========================================================================

# --- 步骤①：ASR 转写（faster-whisper int8，词级时间戳） ---
faster-whisper>=1.0,<2.0

# --- 步骤②：强制对齐（wav2vec2 帧级对齐 → 音素-时间点映射） ---
transformers>=4.45,<5.0
# CPU 版 torch：NAS 无 GPU 时的默认选择，体积约 200MB。
# 有 GPU 的部署机请改装 `torch==<同版本>+cuXXX` 的 CUDA 构建。
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.3,<3.0

# --- 步骤③：音素序列比对（Levenshtein 编辑距离 + 置信度加权） ---
python-Levenshtein>=0.27,<1.0

# --- 步骤④ / §5.4 韵律内部权重：DTW 声学距离 + f0 + 能量 ---

# 能量 RMS：librosa.feature.rms，同上。
dtw-python>=0.2,<0.3

# --- 韵律参考音：学习者 vs 合成参考的 DTW 对齐 ---
piper-tts>=1.3,<2.0

# --- 转写文本规范化（词错误率 WER 计算前的轻规整） ---
unidecode>=1.4,<2.0
